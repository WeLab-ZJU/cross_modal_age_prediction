#!/usr/bin/env bash
# Evaluate a trained FP (fundus) single-modality checkpoint.
# DATA_TAG selects the dataset folder under <HUMAN_DATA_ROOT>/ (dataset | merge).
#
# Single-modality testing: use train_eye.py --eval to load the best checkpoint and evaluate the test set.
# If using another evaluation entry point (such as eval_eye.py), replace the Python command below.

set -e

DATA_TAG="dataset"
MODEL_TYPE="vit_base_patch16"
INPUT_SIZE=256
DATA_TYPE="FP"

ARCH="vit_base_patch16"
EXP_NAME="finetune-HCF-500"

DATA_ROOT="<HUMAN_DATA_ROOT>/${DATA_TAG}/${DATA_TYPE}"
CKPT="<HUMAN_OUTPUT_ROOT>/${DATA_TAG}/${DATA_TYPE}/mae-${ARCH}/eye-${INPUT_SIZE}/${EXP_NAME}/checkpoint-best.pth"
OUTPUT_DIR="<HUMAN_OUTPUT_ROOT>/${DATA_TAG}/${DATA_TYPE}/${ARCH}/eye-${INPUT_SIZE}/${EXP_NAME}"

python <HUMAN_CODE_ROOT>/train_eye.py \
    --eval \
    --batch_size 32 \
    --world_size 1 \
    --model "${ARCH}" \
    --output_dir "${OUTPUT_DIR}" \
    --log_dir "${OUTPUT_DIR}" \
    --nb_classes 4 \
    --data_path "${DATA_ROOT}" \
    --task "eye-${INPUT_SIZE}/finetune-HCF-500" \
    --resume "${CKPT}" \
    --input_size "${INPUT_SIZE}" \
    --class_order "1-30" "30-60" "60-80" "80+"
